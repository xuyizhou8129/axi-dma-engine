# DMA validation test: transfer 1 word from sys_mem to SRAM
#
# Address layout (sys_mem is 1 KB, BRAM is 1 KB):
#   sys_mem 0x0000-0x000F : Descriptor 0 (ring entry 0)
#   sys_mem 0x0040        : Source test data
#   SRAM    0x0100        : DMA destination (byte address)

# --- Step 1: Write source data into sys_mem ---
sysmem_write 00000040 DEADBEEF

# --- Step 2: Write Descriptor 0 into sys_mem at ring base (BASEADDR = 0) ---
# SRC_ADDR = sys_mem 0x40
sysmem_write 00000000 00000040
# DST_ADDR = SRAM byte address 0x100
sysmem_write 00000004 00000100
# LEN = 1 beat
sysmem_write 00000008 00000001
# FLAGS = 1 (DIR=1: System Memory -> SRAM)
sysmem_write 0000000C 00000001

# --- Step 3: Configure DMA ring (do NOT write CTRL here, CMD_RUN_DMA does it) ---
# BASEADDR = 0 (ring starts at sys_mem address 0)
csr_write 00 00000000
# RINGLEN = 2
csr_write 04 00000002
# TAIL = 1 (enqueue descriptor 0)
csr_write 0c 00000001
