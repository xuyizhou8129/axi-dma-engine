# DMA validation test: transfer 4 beats from sys_mem to SRAM
# Matches sim/scenarios/example.csv
#
# Address layout:
#   sys_mem 0x0000-0x000F : Descriptor 0 (ring slot 0)
#   sys_mem 0x0100-0x010F : Source test data (4 words)
#   SRAM    0x0100-0x010F : DMA destination

# --- Step 1: Write source data into sys_mem ---
sysmem_write 00000100 DEADBEEF
sysmem_write 00000104 DEADBEEF
sysmem_write 00000108 DEADBEEF
sysmem_write 0000010C DEADBEEF

# --- Step 2: Write Descriptor 0 into sys_mem at ring base 0x0000 ---
# SRC_ADDR = sys_mem 0x100
sysmem_write 00000000 00000100
# DST_ADDR = SRAM 0x100
sysmem_write 00000004 00000100
# LEN = 4 beats
sysmem_write 00000008 00000004
# FLAGS = 1 (DIR=1: sys_mem -> SRAM)
sysmem_write 0000000C 00000001

# --- Step 3: Configure DMA CSR (CTRL is set by CMD_RUN_DMA, not here) ---
# BASEADDR = 0
csr_write 00 00000000
# RINGLEN = 4
csr_write 04 00000004
# TAIL = 1 (enqueue descriptor 0)
csr_write 0C 00000001
