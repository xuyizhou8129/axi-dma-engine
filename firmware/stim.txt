# DMA validation test: transfer 16 words (64 bytes) from sys_mem to SRAM (DIR=1)
#
# Address layout (sys_mem 1 KB, SRAM 1 KB):
#   sys_mem 0x0000-0x000F : Descriptor 0
#   sys_mem 0x0040-0x007F : Source data (16 words)
#   SRAM    0x0100-0x013F : DMA destination

# --- Step 0: Clear DMA state ---
csr_write 10 00000002
csr_write 10 00000000
csr_write 2c 00000003

# --- Step 1: Source data — 16 words at sys_mem 0x40..0x7C ---
sysmem_write 00000040 10000000
sysmem_write 00000044 10000001
sysmem_write 00000048 10000002
sysmem_write 0000004C 10000003
sysmem_write 00000050 10000004
sysmem_write 00000054 10000005
sysmem_write 00000058 10000006
sysmem_write 0000005C 10000007
sysmem_write 00000060 10000008
sysmem_write 00000064 10000009
sysmem_write 00000068 1000000A
sysmem_write 0000006C 1000000B
sysmem_write 00000070 1000000C
sysmem_write 00000074 1000000D
sysmem_write 00000078 1000000E
sysmem_write 0000007C 1000000F

# --- Step 2: Descriptor 0 (sys_mem 0x40 -> SRAM 0x100, 16 beats, DIR=1) ---
sysmem_write 00000000 00000040
sysmem_write 00000004 00000100
sysmem_write 00000008 00000010
sysmem_write 0000000C 00000001

# --- Step 3: Configure DMA ring ---
csr_write 00 00000000
csr_write 04 00000002
csr_write 0c 00000001
